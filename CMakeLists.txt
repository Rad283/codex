cmake_minimum_required(VERSION 3.24)
project(Orbit LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(MINGW)
  add_link_options(-static -static-libgcc -static-libstdc++)
endif()

include(FetchContent)
set(LUAU_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LUAU_STATIC_CRT ON CACHE BOOL "" FORCE)
FetchContent_Declare(luau
  GIT_REPOSITORY https://github.com/luau-lang/luau.git
  GIT_TAG 0084576e6f8dc9865f27d296f0357794894bd38d
)
FetchContent_MakeAvailable(luau)
set_property(DIRECTORY "${luau_SOURCE_DIR}" PROPERTY EXCLUDE_FROM_ALL TRUE)

add_library(orbit_engine STATIC src/engine.cpp)
target_include_directories(orbit_engine PUBLIC include)
target_link_libraries(orbit_engine PRIVATE Luau.Compiler Luau.VM)
if(MSVC)
  target_compile_options(orbit_engine PRIVATE /W4)
else()
  target_compile_options(orbit_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(engine_tests tests/engine_tests.cpp)
  target_link_libraries(engine_tests PRIVATE orbit_engine)
  add_test(NAME engine COMMAND engine_tests)
  set_tests_properties(engine PROPERTIES TIMEOUT 20)
endif()

if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Orbit supports x64 Windows only.")
  endif()
  add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  add_library(orbit_transport STATIC src/windows/transport.cpp)
  target_include_directories(orbit_transport PUBLIC include)
  add_library(orbit_injection STATIC src/windows/injection.cpp)
  target_link_libraries(orbit_injection PUBLIC orbit_transport)
  add_library(orbit_runtime SHARED src/windows/runtime_dll.cpp)
  target_link_libraries(orbit_runtime PRIVATE orbit_engine orbit_transport advapi32)
  set_target_properties(orbit_runtime PROPERTIES PREFIX "")
  add_executable(orbit_host src/windows/host.cpp)
  add_executable(orbit WIN32 src/windows/app.cpp)
  target_link_libraries(orbit PRIVATE orbit_injection comdlg32 user32 gdi32)
  add_dependencies(orbit orbit_host orbit_runtime)
  if(BUILD_TESTING)
    add_executable(injection_tests tests/injection_tests.cpp)
    target_link_libraries(injection_tests PRIVATE orbit_injection)
    add_dependencies(injection_tests orbit_host orbit_runtime)
    add_test(NAME windows_injection COMMAND injection_tests)
    set_tests_properties(windows_injection PROPERTIES TIMEOUT 45)
    add_executable(ui_tests tests/ui_tests.cpp)
    target_link_libraries(ui_tests PRIVATE orbit_transport user32)
    add_dependencies(ui_tests orbit)
    add_test(NAME windows_ui COMMAND ui_tests)
    set_tests_properties(windows_ui PROPERTIES TIMEOUT 60)
  endif()
endif()
